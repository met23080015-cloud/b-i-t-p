<script>
const studentsEl = document.getElementById('students');
const topicsEl = document.getElementById('topics');
const groupCountEl = document.getElementById('groupCount');
const drawBtn = document.getElementById('drawBtn');
const redrawBtn = document.getElementById('redrawBtn');
const warnBox = document.getElementById('warnBox');
const emptyState = document.getElementById('emptyState');
const ticketGrid = document.getElementById('ticketGrid');
const summaryLine = document.getElementById('summaryLine');
const toolbar = document.getElementById('toolbar');
const stuCount = document.getElementById('stuCount');
const topicCount = document.getElementById('topicCount');

let lastResult = null;

function parseLines(text){
  return text.split('\n').map(s => s.trim()).filter(Boolean);
}

function updateCounts(){
  stuCount.textContent = parseLines(studentsEl.value).length + ' người';
  topicCount.textContent = parseLines(topicsEl.value).length + ' đề tài';
}
studentsEl.addEventListener('input', updateCounts);
topicsEl.addEventListener('input', updateCounts);
updateCounts();

function shuffle(arr){
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--){
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function showWarn(msg){
  if (!msg){ warnBox.style.display = 'none'; warnBox.textContent = ''; return; }
  warnBox.style.display = 'block';
  warnBox.textContent = msg;
}

function draw(){
  const students = parseLines(studentsEl.value);
  const topicsRaw = parseLines(topicsEl.value);
  const n = parseInt(groupCountEl.value, 10);

  if (students.length < 2){
    showWarn('Cần ít nhất 2 thành viên để chia nhóm.');
    return;
  }
  if (!n || n < 2){
    showWarn('Số nhóm phải là số nguyên từ 2 trở lên.');
    return;
  }
  if (n > students.length){
    showWarn('Số nhóm không thể nhiều hơn số thành viên.');
    return;
  }
  showWarn('');

  const shuffledStudents = shuffle(students);
  const groups = Array.from({length:n}, () => []);
  shuffledStudents.forEach((name, i) => groups[i % n].push(name));

  let topicAssignment = [];
  let topicNote = '';
  if (topicsRaw.length > 0){
    const shuffledTopics = shuffle(topicsRaw);
    if (shuffledTopics.length < n){
      topicNote = 'Số đề tài (' + shuffledTopics.length + ') ít hơn số nhóm (' + n + ') nên một vài đề tài sẽ lặp lại.';
    }
    for (let i = 0; i < n; i++){
      topicAssignment.push(shuffledTopics[i % shuffledTopics.length]);
    }
    if (topicNote) showWarn(topicNote);
  }

  lastResult = { groups, topicAssignment, ts: new Date() };
  render(lastResult);
}

function render(result){
  const { groups, topicAssignment } = result;
  emptyState.style.display = 'none';
  toolbar.style.display = 'flex';
  ticketGrid.innerHTML = '';

  const totalPeople = groups.reduce((s,g) => s + g.length, 0);
  summaryLine.innerHTML = 'Đã chia <b>' + totalPeople + '</b> người vào <b>' + groups.length + '</b> nhóm';

  groups.forEach((members, idx) => {
    const num = String(idx + 1).padStart(2, '0');
    const topic = topicAssignment[idx];
    const card = document.createElement('div');
    card.className = 'ticket';
    card.style.animationDelay = (idx * 0.05) + 's';
    card.innerHTML =
      '<div class="stub">' +
        '<span class="grpnum">Nhóm ' + num + '</span>' +
        '<span class="serial">#' + (Math.random().toString(36).slice(2,7).toUpperCase()) + '</span>' +
      '</div>' +
      '<div class="perforation"></div>' +
      '<div class="body">' +
        '<h3>Thành viên (' + members.length + ')</h3>' +
        '<ul>' + members.map(m => '<li>' + escapeHtml(m) + '</li>').join('') + '</ul>' +
        (topic ? '<div class="stamp"><b>Đề tài bốc được</b>' + escapeHtml(topic) + '</div>' : '') +
      '</div>';
    ticketGrid.appendChild(card);
  });
}

function escapeHtml(s){
  const d = document.createElement('div');
  d.textContent = s;
  return d.innerHTML;
}

function resultAsText(){
  if (!lastResult) return '';
  const lines = ['KẾT QUẢ BỐC THĂM CHIA NHÓM', '===================================', ''];
  lastResult.groups.forEach((members, idx) => {
    lines.push('Nhóm ' + (idx + 1) + ' (' + members.length + ' người)');
    members.forEach(m => lines.push('  - ' + m));
    if (lastResult.topicAssignment[idx]) lines.push('  Đề tài: ' + lastResult.topicAssignment[idx]);
    lines.push('');
  });
  return lines.join('\n');
}

drawBtn.addEventListener('click', () => {
  draw();
  drawBtn.style.display = 'none';
  redrawBtn.style.display = 'block';
});
redrawBtn.addEventListener('click', draw);

document.getElementById('copyBtn').addEventListener('click', () => {
  navigator.clipboard.writeText(resultAsText()).then(() => {
    const btn = document.getElementById('copyBtn');
    const old = btn.textContent;
    btn.textContent = 'Đã sao chép';
    setTimeout(() => btn.textContent = old, 1400);
  });
});

document.getElementById('downloadBtn').addEventListener('click', () => {
  const blob = new Blob([resultAsText()], { type: 'text/plain;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'ket-qua-chia-nhom.txt';
  a.click();
  URL.revokeObjectURL(url);
});
</script>
